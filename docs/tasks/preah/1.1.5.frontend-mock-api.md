### 1. Axios 설정 및 API 베이스 URL 구성
위치: `frontend/src/services/api.ts`

```typescript
import axios from 'axios';

// Axios 인스턴스 생성
const api = axios.create({
  baseURL: process.env.NEXT_PUBLIC_API_URL || 'http://localhost:8080/api',
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// 요청 인터셉터
api.interceptors.request.use(
  (config) => {
    // 개발 환경에서 요청 로깅
    if (process.env.NODE_ENV === 'development') {
      console.log('API Request:', config);
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// 응답 인터셉터
api.interceptors.response.use(
  (response) => {
    return response;
  },
  (error) => {
    // 백엔드가 없을 때 Mock 데이터 반환
    if (error.code === 'ECONNREFUSED' || error.code === 'ERR_NETWORK') {
      console.log('Backend unavailable, using mock data');
      return Promise.resolve({
        data: getMockData(error.config.url),
        status: 200,
        statusText: 'OK (Mock)',
        headers: {},
        config: error.config,
      });
    }
    return Promise.reject(error);
  }
);

export default api;
```

### 2. Mock Health Check 응답 구현
위치: `frontend/src/services/mockData.ts`

```typescript
// Mock 데이터 저장소
const mockResponses: Record<string, any> = {
  '/health': {
    status: 'UP',
    database: 'UP',
    timestamp: new Date().toISOString(),
  },
  '/todos': [
    {
      id: 1,
      title: 'Mock 할일 1',
      completed: false,
      createdAt: new Date().toISOString(),
    },
    {
      id: 2,
      title: 'Mock 할일 2',
      completed: true,
      createdAt: new Date().toISOString(),
    },
  ],
};

export const getMockData = (url: string): any => {
  // URL에서 엔드포인트 추출
  const endpoint = url.replace(process.env.NEXT_PUBLIC_API_URL || '', '');

  // Mock 데이터 반환
  if (mockResponses[endpoint]) {
    return mockResponses[endpoint];
  }

  // 기본 응답
  return {
    status: 'UP',
    message: 'Mock response',
  };
};

// Health Check 서비스
export const healthCheckService = {
  async check() {
    try {
      const response = await api.get('/health');
      return response.data;
    } catch (error) {
      console.error('Health check failed:', error);
      // Mock 데이터가 이미 인터셉터에서 처리됨
      throw error;
    }
  },
};
```

### 3. 환영 페이지에 API 상태 표시
위치: `frontend/src/app/page.tsx`

```typescript
'use client';

import { useState, useEffect } from 'react';
import { Box, Container, Typography, Paper, Chip, Alert } from '@mui/material';
import { Spa } from '@mui/icons-material';
import { healthCheckService } from '@/services/mockData';

interface HealthStatus {
  status: string;
  database: string;
  timestamp: string;
}

export default function Home() {
  const [healthStatus, setHealthStatus] = useState<HealthStatus | null>(null);
  const [isMockData, setIsMockData] = useState(false);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    const checkHealth = async () => {
      try {
        const response = await healthCheckService.check();
        setHealthStatus(response);

        // Mock 데이터 여부 확인
        if (response.statusText === 'OK (Mock)') {
          setIsMockData(true);
        }
        setError(null);
      } catch (err) {
        setError('서버 연결에 실패했습니다');
        // Mock 데이터로 폴백
        setHealthStatus({
          status: 'UP',
          database: 'MOCK',
          timestamp: new Date().toISOString(),
        });
        setIsMockData(true);
      }
    };

    // 초기 체크
    checkHealth();

    // 10초마다 체크
    const interval = setInterval(checkHealth, 10000);

    return () => clearInterval(interval);
  }, []);

  const getStatusColor = () => {
    if (error) return 'error';
    if (healthStatus?.status === 'UP' && healthStatus?.database === 'UP') {
      return 'success';
    }
    return 'warning';
  };

  const getStatusText = () => {
    if (isMockData) return 'Mock 데이터 사용 중';
    if (error) return '서버 연결 실패';
    if (healthStatus?.status === 'UP' && healthStatus?.database === 'UP') {
      return '정상 작동 중';
    }
    return '부분 작동';
  };

  return (
    <Container maxWidth="md">
      <Box sx={{
        minHeight: '100vh',
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'center',
        flexDirection: 'column',
        gap: 2
      }}>
        {/* API 상태 표시 */}
        <Box sx={{ position: 'fixed', top: 16, right: 16 }}>
          <Chip
            label={getStatusText()}
            color={getStatusColor()}
            variant="filled"
            size="small"
          />
        </Box>

        {/* Mock 데이터 알림 */}
        {isMockData && (
          <Alert severity="info" sx={{ width: '100%', maxWidth: 600 }}>
            백엔드 서버가 실행되지 않아 Mock 데이터를 사용하고 있습니다.
            실제 데이터를 보려면 백엔드 서버를 시작해주세요.
          </Alert>
        )}

        {/* 메인 컨텐츠 */}
        <Paper elevation={3} sx={{ p: 6, textAlign: 'center' }}>
          <Spa sx={{ fontSize: 80, color: 'green', mb: 2 }} />
          <Typography variant="h3" gutterBottom>
            감성적인 할일 목록
          </Typography>
          <Typography variant="h6" color="text.secondary" gutterBottom>
            작은 성취가 주는 큰 위로
          </Typography>
          <Typography variant="body1" sx={{ mt: 3 }}>
            곧 당신의 바질 화분이 자라날 준비를 하고 있어요 🌱
          </Typography>
        </Paper>

        {/* 디버그 정보 (개발 환경에서만) */}
        {process.env.NODE_ENV === 'development' && healthStatus && (
          <Paper elevation={1} sx={{ p: 2, mt: 2, maxWidth: 600 }}>
            <Typography variant="caption" component="pre">
              {JSON.stringify(healthStatus, null, 2)}
            </Typography>
          </Paper>
        )}
      </Box>
    </Container>
  );
}
```
